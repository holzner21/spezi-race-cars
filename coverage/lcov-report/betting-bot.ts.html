    betConfig: any,
    maxRetries: number = 5,
    initialDelayMs: number = 5000
  ): Promise<any> {
    let lastError: Error | null = null;

          } else {
            console.log(chalk.red(`❌ API throttled after ${maxRetries} retries. Stopping.`));
            throw new Error('Maximum retries exceeded due to API throttling');

    }
